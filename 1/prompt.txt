const titles = [
  "Start with your website.",
  "Where’s your visitor data?",
  "Give your agent a place to work.",
  "Ready for less friction.",
];
const descriptions = [
  "One website. One agent. A continuous loop of better.",
  "Choose the analytics platform you use.",
  "Choose the GitHub repository for this website.",
  "Pick a goal and review your setup.",
];
function Onboarding() {
  const workspace = useWorkspace();
  const navigate = useNavigate();
  const [step, setStep] = useState(0);
  const [site, setSite] = useState("");
  const [analytics, setAnalytics] = useState("PostHog");
  const [repo, setRepo] = useState("");
  const [goal, setGoal] = useState("Signup conversion");
  const [error, setError] = useState("");
  const next = () => {
    setError("");
    if (step === 0) {
      try {
        const url = new URL(site.includes("://") ? site : `https://${site}`);
        if (
          !url.hostname.includes(".") ||
          !["http:", "https:"].includes(url.protocol)
        )
          throw new Error();
        setSite(url.origin);
      } catch {
        setError("Enter a valid website address, like acme.so.");
        return;
      }
    }
    if (step === 2 && !/^[\w.-]+\/[\w.-]+$/.test(repo)) {
      setError("Enter a repository as owner/repository.");
      return;
    }
    if (step < 3) {
      setStep(step + 1);
      return;
    }
    workspace.setSite(site);
    workspace.setAnalytics(analytics);
    workspace.setRepo(repo);
    workspace.setGoal(goal);
    workspace.setPaused(false);
    navigate({ to: "/dashboard" });
  };
  return (
    <div className="onboarding-page">
      <header className="container-page onboarding-header">
        <ProductBrand />
        <div className="onboarding-header-actions">
          <Button variant="ghost" asChild>
            <Link to="/dashboard">
              Go to dashboard
              <ArrowRight />
            </Link>
          </Button>
        </div>
      </header>
      <main className="onboarding-main">
        <div
          className="onboarding-progress"
          aria-label={`Step ${step + 1} of 4`}
        >
          {titles.map((title, i) => (
            <span key={title} className={i <= step ? "complete" : ""} />
          ))}
        </div>
        <div className="eyebrow text-muted-foreground">
          0{step + 1} / 04 —{" "}
          {["YOUR WEBSITE", "ANALYTICS", "REPOSITORY", "YOUR GOAL"][step]}
        </div>
        <h1>{titles[step]}</h1>
        <p className="onboarding-intro">{descriptions[step]}</p>
        <form
          onSubmit={(event) => {
            event.preventDefault();
            next();
          }}
        >
          {step === 0 && (
            <label className="field">
              Website URL
              <input
                autoFocus
                className="product-input"
                type="text"
                inputMode="url"
                placeholder="https://yourwebsite.com"
                value={site}
                onChange={(event) => setSite(event.target.value)}
              />
              <span className="field-hint">
                The website you want to improve.
              </span>
            </label>
          )}
          {step === 1 && (
            <div className="onboarding-choices">
              {[
                { name: "PostHog", icon: ChartNoAxesCombined },
                { name: "Mixpanel", icon: ChartNoAxesCombined },
                { name: "Amplitude", icon: Layers },
              ].map((provider) => (
                <Button
                  key={provider.name}
                  type="button"
                  variant="outline"
                  aria-pressed={analytics === provider.name}
                  className={`onboarding-choice ${analytics === provider.name ? "selected" : ""}`}
                  onClick={() => setAnalytics(provider.name)}
                >
                  <provider.icon />
                  {provider.name}
                  {analytics === provider.name && <Check />}
                </Button>
              ))}
            </div>
          )}
          {step === 2 && (
            <>
              <label className="field">
                <span className="flex items-center gap-2">
                  <Github size={15} />
                  GitHub repository
                </span>
                <input
                  autoFocus
                  className="product-input"
                  placeholder="your-team/website"
                  value={repo}
                  onChange={(event) => setRepo(event.target.value)}
                />
                <span className="field-hint">
                  Use the repository that powers this website.
                </span>
              </label>
              <div className="setting-row">
                <div>
                  <strong>Review changes before they go live</strong>
                  <p>You approve. The agent measures.</p>
                </div>
                <Check size={18} className="text-success" />
              </div>
            </>
          )}
          {step === 3 && (
            <>
              <label className="field">
                Primary goal
                <select
                  className="product-input"
                  value={goal}
                  onChange={(event) => setGoal(event.target.value)}
                >
                  <option>Signup conversion</option>
                  <option>Checkout completion</option>
                  <option>Contact form submissions</option>
                </select>
              </label>
              <div className="onboarding-review">
                <div>
                  <span>Website</span>
                  <strong>{site}</strong>
                </div>
                <div>
                  <span>Analytics</span>
                  <strong>{analytics}</strong>
                </div>
                <div>
                  <span>Repository</span>
                  <strong>{repo}</strong>
                </div>
                <div>
                  <span>Change approval</span>
                  <strong>Review before publishing</strong>
                </div>
              </div>
            </>
          )}
          {error && (
            <p className="form-error" role="alert">
              {error}
            </p>
          )}
          <div className="onboarding-actions">
            {step > 0 ? (
              <Button
                variant="ghost"
                type="button"
                onClick={() => {
                  setStep(step - 1);
                  setError("");
                }}
              >
                <ArrowLeft />
                Back
              </Button>
            ) : (
              <span className="field-hint">About a minute.</span>
            )}
            <Button variant="hero" type="submit">
              {step === 3 ? "Start my agent" : "Continue"}
              {step === 3 ? <Sparkles /> : <ArrowRight />}
            </Button>
          </div>
        </form>
      </main>
    </div>
  );
}
